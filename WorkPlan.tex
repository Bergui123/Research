\section{Goal}
% =====================================================================

\textbf{Research question.} \emph{Fault-Tolerant Resource Estimation for
Schr\"odingerised Hamilton--Jacobi Equations: Logical Qubit and T-Gate Counts
for Multi-Asset Optimal Execution.}

\textbf{Deliverable.} A formula and a table giving the number of logical qubits
$N_q$ and T-gates $N_T$ needed to solve the two-asset optimal execution problem on
a fault-tolerant quantum computer, as functions of the grid size, the target
precision $\varepsilon$, the horizon $T$, and the Almgren--Chriss parameters
($\lambda$, the covariance matrix $C$, the impact matrix $H$). Every parameter in
the count must be traceable to a numbered equation in Almgren and Chriss.

\textbf{What the paper does not claim.} It does not propose a new trading
strategy, and it does not claim a quantum speedup. For this linear-quadratic
problem the classical solution is a closed-form Riccati ODE. The contribution is
the cost of running the full pipeline on a benchmark instance.

\textbf{The chain.}
\begin{center}
\begin{tabular}{@{}c@{}}
Almgren--Chriss objective $\;\to\;$ integral $\;\to\;$ Hamilton--Jacobi PDE\\[0.3em]
$\to\;$ linear equation (Cole--Hopf) $\;\to\;$ Schr\"odinger form\\[0.3em]
$\to\;$ qubit and T-gate count
\end{tabular}
\end{center}

\subsection*{Stage 2 --- Lock the classical benchmark}
% ---------------------------------------------------------------------

\textbf{Do.} Record the following as the reference solution that every later stage
is checked against. With $H=\mathrm{diag}(\eta_j)$ and $\dot{\mathbf x}=-\mathbf v$:

\begin{align}
  J &= \int_0^T\!\big(\mathbf v^\top H\mathbf v+\lambda\,\mathbf x^\top C\mathbf x\big)dt \\
  \partial_t\mathcal V &-\tfrac14(\nabla\mathcal V)^\top H^{-1}\nabla\mathcal V
      +\lambda\,\mathbf x^\top C\mathbf x=0 \label{eq:pde}\\
  \mathcal V &=\tfrac12\mathbf x^\top P(t)\mathbf x,\qquad
      \dot P=\tfrac12PH^{-1}P-2\lambda C \label{eq:ric}\\
  \mathbf v^* &=\tfrac12H^{-1}P\,\mathbf x \\
  \mathbf x(t) &= H^{-1/2}U\,\mathrm{diag}\!\Big(\frac{\sinh\kappa_i(T-t)}
      {\sinh\kappa_iT}\Big)U^\top H^{1/2}\mathbf X,\qquad
      \kappa_i^2=\lambda\mu_i,\ \ H^{-1/2}CH^{-1/2}=U\,\mathrm{diag}(\mu_i)U^\top
\end{align}
and the optimal cost $\mathcal V(\mathbf X,0)=\tfrac12\mathbf X^\top P(0)\mathbf X$.

\textbf{Source.} Derived in these notes using dynamic programming and the standard
linear--quadratic (Riccati) method; consistent with Almgren--Chriss Eqs.~16--17
and 31 in the limits checked.\\
\textbf{Exit criterion.} For one asset the trajectory equals
$X\sinh(\kappa(T-t))/\sinh(\kappa T)$ (met); the two-asset case reproduces the
direct minimisation of the discrete cost (met for one parameter set).

\end{enumerate}
\textbf{Source.} Your choice; market calibration is outside the paper's scope.\\
\textbf{Exit criterion.} A parameter table with all checks passed.\\
\textbf{Suggested owner.} Project lead.

% ---------------------------------------------------------------------
\subsection*{Stage 4 --- Time-to-go form and Cole--Hopf (linearisation)}
% ---------------------------------------------------------------------

\textbf{Do.} Reverse time ($s=T-t$, $W(\mathbf x,s)=\mathcal V(\mathbf x,T-s)$) and add a viscosity
$\nu>0$:
\begin{equation}
  \partial_sW+\tfrac14\nabla W^\top H^{-1}\nabla W=\lambda\,\mathbf x^\top C\mathbf x
    +\nu\,\nabla\!\cdot\!(H^{-1}\nabla W),\qquad W(\mathbf x,0)=\tfrac12\mathbf x^\top\Phi\mathbf x
\end{equation}
Substitute $\psi=e^{-W/(4\nu)}$. The nonlinear term cancels:
\begin{equation}
  \partial_s\psi=-\hat H\psi,\qquad
  \hat H=-\nu\,\nabla\!\cdot\!(H^{-1}\nabla)+\frac{\lambda}{4\nu}\,\mathbf x^\top C\mathbf x
\end{equation}
Recover the trading rate from $\mathbf v^*=-2\nu\,H^{-1}\nabla\ln\psi$.

\textbf{Key claim to verify.} Writing $W=\tfrac12\mathbf x^\top P(s)\mathbf x+c(s)$, the
quadratic part obeys the same Riccati equation as~\eqref{eq:ric}, and
$\dot c=\nu\,\mathrm{tr}(H^{-1}P)$. So $\nu$ changes only an additive constant, not
the gradient, and $\nu$ need not tend to zero for this problem.\\
\textbf{Source.} The Hopf--Cole transform is standard (Evans, \emph{Partial
Differential Equations}); this matrix version is derived in these notes and is
\textbf{not yet checked}.\\
\textbf{Exit criterion.} Substituting $\psi=e^{-W/(4\nu)}$ back by hand reproduces
the equation; for one asset the gradient from $\psi$ equals the Stage~2 gradient.\\
\textbf{Suggested owner.} Math/control lead.

% ---------------------------------------------------------------------
\subsection*{Stage 5 --- Spectrum and length scales; choosing $\nu$}
% ---------------------------------------------------------------------

\textbf{Do.} With $\mathbf y=U^\top H^{1/2}\mathbf x$, the operator $\hat H$ splits into
independent oscillators:
\begin{equation}
  \hat H=\sum_i\Big[-\nu\,\partial_{y_i}^2+\frac{\lambda\mu_i}{4\nu}y_i^2\Big],\qquad
  E_{\mathbf n}=\sum_i\kappa_i\big(n_i+\tfrac12\big),\qquad
  \ell_i=\sqrt{2\nu/\kappa_i}
\end{equation}
The spectrum does not depend on $\nu$; the ground-state width $\ell_i$ does. The norm
of $\psi$ decays like $e^{-E_0T}$ with $E_0=\tfrac12\sum_i\kappa_i$.

\textbf{Decide.} $\nu$ trades two things: $\psi$ at the starting inventory
$\mathbf X$ has size about $e^{-W(\mathbf X)/(4\nu)}$ (small $\nu$ makes it tiny and
the readout expensive), while $\ell_i$ sets how fine the grid must be (large $\nu$
widens the state and relaxes the grid). Sweep $\nu$.\\
\textbf{Source.} Derived here from the standard harmonic oscillator; unchecked.\\
\textbf{Exit criterion.} A table of $\ell_i$, $E_0T$, and $\psi(\mathbf X)$ versus $\nu$
for the Stage~3 instance.

% ---------------------------------------------------------------------
\subsection*{Stage 6 --- Discretisation}
% ---------------------------------------------------------------------

\textbf{Do.} Use $N_x=2^{n_x}$ grid points per asset on $[-L,L]$, spacing $\Delta x$,
with $L\ge\max_j X_j$ and $\Delta x$ small compared with $\min_i\ell_i$. The discrete
operator is
\begin{equation}
  \hat H_d=\nu\sum_a\frac1{\eta_a}(-D_{2,a})+\frac{\lambda}{4\nu}\sum_{a,b}C_{ab}X_aX_b,
  \qquad
  \|\hat H_d\|\lesssim\frac{4\nu}{\Delta x^2}\sum_a\frac1{\eta_a}
     +\frac{\lambda L^2}{4\nu}\sum_{a,b}|C_{ab}|
\end{equation}
where $D_{2,a}$ is the second-difference matrix for asset $a$. The norm bound is an
estimate from these notes.\\
\textbf{Exit criterion.} The error in the gradient at $\mathbf x=\mathbf X$ is
quantified as a function of $\Delta x$, $L$, $\nu$, and compared with the exact
Gaussian solution from Stage~5.\\
\textbf{Suggested owner.} Numerics lead.

% ---------------------------------------------------------------------
\subsection*{Stage 7 --- Schr\"odingerisation}
% ---------------------------------------------------------------------

\textbf{First read} the Jin--Liu--Yu papers (Section~4). The sketch below is from
memory and must be replaced by their equations and conditions.

\textbf{Sketch.} $A=-\hat H_d$ is Hermitian and negative semidefinite. Set
$w=e^{-p}\psi$ for $p>0$ (extended to $p<0$), and Fourier-transform in $p$ with
frequencies $\eta_k$. The result is a Schr\"odinger equation
\begin{equation}
  i\,\partial_s\hat w=H_S\hat w,\qquad H_S=-\hat H_d\otimes D_\eta,\qquad
  \|H_S\|=\|\hat H_d\|\,\eta_{\max},\ \ \eta_{\max}=\pi/\Delta p
\end{equation}
and $\psi$ is recovered as $e^{p^*}w(\cdot,p^*)$ for any $p^*>0$.

\textbf{Derive from the papers.} The conditions on the $p$-domain and $\Delta p$; the
success probability of the recovery step, including the factor from the norm decay
$e^{-E_0T}$ (a known constant shift of $\hat H$ removes it and does not change the
gradient).\\
\textbf{Exit criterion.} A small classical simulation of the Schr\"odingerised
system recovers $\psi$ to precision $\varepsilon$.\\
\textbf{Suggested owner.} Math lead with the resource-estimation lead.

% ---------------------------------------------------------------------
\subsection*{Stage 8 --- Resource count}
% ---------------------------------------------------------------------

\textbf{Do.}
\begin{enumerate}[leftmargin=1.5em,nosep]
  \item Qubits: $N_q=M\,n_x+n_p+n_{\mathrm{anc}}$, with $n_p=\log_2N_p$ and
        $n_{\mathrm{anc}}$ from the block-encoding.
  \item Queries to the block-encoding: $Q=O\!\big(\alpha T+\log(1/\varepsilon)/\log\log(1/\varepsilon)\big)$
        with $\alpha\ge\|H_S\|$ (qubitization or QSVT-style Hamiltonian simulation).
        Take explicit constants from the primary simulation papers.
  \item T-gates: $N_T=Q\times(\text{T-cost of one block-encoding})$ plus rotation
        synthesis. The kinetic term costs adders on $n_x$-bit registers; the
        potential term $x_ax_b$ costs $O(n_x^2)$ Toffoli gates per product.
  \item Cross-check the analytic count with a circuit-level estimate from a
        resource-estimation tool.
\end{enumerate}
\textbf{Source.} Qubitization and QSVT literature (Low--Chuang; Gily\'en et al.),
cited from memory and not yet opened.\\
\textbf{Exit criterion.} Formulas for $N_q$ and $N_T$, plus a circuit-level number
for the Stage~3 instance that agrees with them within the stated slack.\\
\textbf{Suggested owner.} Resource-estimation lead with the software/circuit engineer.

% ---------------------------------------------------------------------
\subsection*{Stage 9 --- Readout}
% ---------------------------------------------------------------------

\textbf{Decide} what is extracted: the trading rate $\mathbf v^*(\mathbf X,0)$ needs
$\nabla\ln\psi$ at $\mathbf x=\mathbf X$, for example from the amplitudes at
neighbouring grid points by amplitude estimation. State its cost explicitly and add it
to $N_T$; it may dominate.\\
\textbf{Exit criterion.} The readout cost is a formula in $\varepsilon$ and in
$\psi(\mathbf X)$ from Stage~5.

% ---------------------------------------------------------------------
\subsection*{Stage 10 --- Validation, sweeps, write-up}
% ---------------------------------------------------------------------

\begin{enumerate}[leftmargin=1.5em,nosep]
  \item Validate every stage against the Stage~2 benchmark; the $\lambda\to0$ limit
        gives the straight-line schedule and large $\lambda$ gives fast liquidation.
  \item Sweep $\lambda$, $\rho$, $\varepsilon$, $\nu$, and $n_x$; report the
        resulting $N_q$ and $N_T$ as curves, not single numbers.
  \item Write the formulas with $M$ explicit and report $M=2$ as an instance, so the
        scaling in $M$ is visible.
  \item Cite every equation to its source; mark equations derived here as such.
\end{enumerate}
\textbf{Suggested owner.} Co-writing lead with the project lead.

\section{Risks}
% =====================================================================

\begin{itemize}[leftmargin=1.5em]
  \item \textbf{Cole--Hopf claim unchecked.} Everything after Stage~4 rests on it.
  \item \textbf{Schr\"odingerisation steps from memory.} The primary papers must be
        read before Stage~7.
  \item \textbf{Prior work.} The closest paper found (Schr\"odingerization with
        Hamilton--Jacobi-type problems) has not been read. No claim of novelty should
        be made before it is.
  \item \textbf{Stiffness.} A large terminal penalty $\Phi$ affects conditioning and
        therefore the gate count.
  \item \textbf{Readout cost} may dominate the total.
  \item \textbf{Classical triviality.} A reviewer will point out the closed-form
        solution; the framing in Section~1 answers this.
\end{itemize}
